How to Replicate This in R with OlinkAnalyze

Below is the complete, reproducible R code to load, QC, normalize, and run paired linear mixed models on real Olink NPX data: 

library(OlinkAnalyze)
library(tidyverse)

# 1. Read Olink Export (CSV or Parquet for Explore HT)
df <- read_NPX("olink_cohort_npx.csv")

# 2. Sample Quality Control
# Flag wells where internal controls deviated > +/- 0.3 NPX
df_clean <- df %>%
  filter(QC_Warning == "Pass")

# 3. Assay Detectability Filter (Keep proteins detected in >= 50% of samples)
passing_assays <- df_clean %>%
  group_by(Assay) %>%
  summarise(Pct_Above_LOD = mean(NPX > LOD, na.rm = TRUE) * 100) %>%
  filter(Pct_Above_LOD >= 50) %>%
  pull(Assay)

df_filtered <- df_clean %>% filter(Assay %in% passing_assays)

# 4. Multi-Plate Bridging Normalization
df_norm <- olink_normalization(
  df = df_filtered,
  overlapping_samples_df1 = df_filtered %>% filter(grepl("BRIDGE", SampleID)),
  df1_project_nr = "Plate1",
  df2_project_nr = "Plate2",
  reference_project = "Plate1"
)

# 5. Paired Differential Abundance Testing (Post vs Pre)
# Using paired t-test:
paired_results <- olink_ttest(
  df = df_norm,
  variable = "Timepoint", # e.g., Pre vs Post
  pair_id = "PatientID"   # Enforces paired analysis!
)

# Or using Linear Mixed-Effects Model with Covariates:
lmer_results <- olink_lmer(
  df = df_norm,
  variable = "Timepoint",
  random = "PatientID",
  covariates = c("Age", "Sex")
)
